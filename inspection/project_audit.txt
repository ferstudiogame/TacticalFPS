===== FILE: README.md =====
# Tactical FPS for Android

Proyecto prototipo FPS táctico móvil construido con Godot Engine 4.7.2.

## Compilación automática

El repositorio conserva el código fuente comprimido en `TacticalFPS_Godot472_source.zip`. El workflow de GitHub Actions extrae el proyecto, importa recursos, ejecuta validaciones headless y exporta un APK Android ARM64 firmado con una clave debug temporal.

1. Abre la pestaña **Actions** del repositorio.
2. Entra en **Build Tactical FPS Android APK**.
3. Revisa los pasos de validación y exportación.
4. Descarga el APK desde **Artifacts** o desde **Releases** si el workflow termina correctamente.

## Nota sobre la firma

El APK se firma con una clave debug generada durante cada ejecución. Sirve para instalación y pruebas, pero no es una firma de publicación estable para Google Play.

## Estado

El proyecto se considera validado solo cuando una ejecución real de GitHub Actions termina correctamente y produce el APK. Un commit o la mera presencia del workflow no demuestran que el juego funcione.

func configure(start_point: Vector3, end_point: Vector3) -> void:
	var direction: Vector3 = end_point - start_point
	var distance: float = maxf(direction.length(), 0.05)
	var beam: BoxMesh = BoxMesh.new()
	beam.size = Vector3(0.012, 0.012, distance)
	mesh = beam
	var material: StandardMaterial3D = StandardMaterial3D.new()

	material.albedo_color = Color(1.0, 0.78, 0.32, 0.82)
	material.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
	material.emission_enabled = true
	material.emission = Color(1.0, 0.52, 0.12)
	material.emission_energy_multiplier = 1.5
	material.no_depth_test = false
	material_override = material
	global_position = (start_point + end_point) * 0.5
	if direction.length_squared() > 0.0001:
		look_at(end_point, Vector3.UP)
	cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	_life_left = life_seconds

func _process(delta: float) -> void:
	_life_left -= delta
	if _life_left <= 0.0:
		queue_free()

func _physics_process(delta: float) -> void:
	if not _alive:
		return
	var patrol_offset: float = position.x - _patrol_origin_x
	if patrol_offset > 1.8:
		_patrol_direction = -1.0
	elif patrol_offset < -1.8:
		_patrol_direction = 1.0
	_reaction_velocity = _reaction_velocity.lerp(Vector3.ZERO, clampf(delta * 4.5, 0.0, 1.0))
	position.x += (_patrol_direction * patrol_speed + _reaction_velocity.x) * delta
	position.z += _reaction_velocity.z * delta
	_health_display_left = maxf(_health_display_left - delta, 0.0)
	if _health_bar_root != null:
		_health_bar_root.visible = _health_display_left > 0.0

func take_damage(amount: float, headshot: bool) -> bool:
	if not _alive:
		return false
	current_health = maxf(current_health - amount, 0.0)
	_reaction_velocity += Vector3(0.0, 0.0, 1.1 if headshot else 0.55)
	_health_display_left = 1.6
	_update_health_display()
	if headshot:
		_play_headshot_sound()
	if current_health <= 0.0:
		_die()
		return true
	return false

func _die() -> void:
	_alive = false
	visible = false
	if _head_area != null:
		_head_area.monitorable = false
	if _torso_area != null:
		_torso_area.monitorable = false
	_respawn_later()

func _respawn_later() -> void:
	await get_tree().create_timer(respawn_delay).timeout
	if not is_inside_tree():
		return
	current_health = max_health
	_alive = true
	position.x = _patrol_origin_x
	position.z = _patrol_origin_z
	visible = true
	if _head_area != null:
		_head_area.monitorable = true
	if _torso_area != null:
		_torso_area.monitorable = true
	_update_health_display()

func _physics_process(delta: float) -> void:
	_slide_cooldown_left = maxf(_slide_cooldown_left - delta, 0.0)
	var input_vector: Vector2 = _get_move_input()
	var move_direction: Vector3 = (global_transform.basis * Vector3(input_vector.x, 0.0, input_vector.y)).normalized()
	var sprint_requested: bool = Input.is_action_pressed("sprint") or _touch_tactical_sprint
	var moving: bool = input_vector.length_squared() > 0.035
	var sprinting: bool = sprint_requested and moving and not _is_crouched and not _is_sliding
	var tactical_sprinting: bool = sprinting and (_touch_tactical_sprint or Input.is_action_pressed("tactical_sprint"))

	if Input.is_action_just_pressed("crouch"):
		request_crouch()
	if _touch_jump or Input.is_action_just_pressed("jump"):
		request_jump()
		_touch_jump = false
	if Input.is_action_just_pressed("reload"):
		request_reload()

	if _is_sliding:
		_slide_left -= delta
		var slide_strength: float = clampf(_slide_left / slide_duration, 0.0, 1.0)
		var slide_velocity: Vector3 = _slide_direction * lerpf(2.0, slide_speed, slide_strength)
		velocity.x = slide_velocity.x
		velocity.z = slide_velocity.z
		if _slide_left <= 0.0:
			_is_sliding = false
			_is_crouched = true
	elif is_on_floor():
		var target_speed: float = walk_speed
		if _is_crouched:
			target_speed = crouch_speed
		elif tactical_sprinting:
			target_speed = tactical_sprint_speed
		elif sprinting:
			target_speed = sprint_speed
		var target_velocity: Vector3 = move_direction * target_speed
		var rate: float = acceleration if moving else acceleration * 1.35
		velocity.x = move_toward(velocity.x, target_velocity.x, rate * delta)
		velocity.z = move_toward(velocity.z, target_velocity.z, rate * delta)
	else:
		velocity.x = move_toward(velocity.x, move_direction.x * walk_speed, air_control * delta)
		velocity.z = move_toward(velocity.z, move_direction.z * walk_speed, air_control * delta)

	if not is_on_floor():
		velocity.y -= 9.8 * delta
	else:
		if velocity.y < 0.0:
			velocity.y = -0.5
	move_and_slide()
	_update_camera(delta, input_vector, moving, sprinting, tactical_sprinting)
	weapon.set_sprinting(tactical_sprinting, _is_sliding)
	weapon.set_fire_pressed(_touch_fire or Input.is_action_pressed("fire"))
	weapon.set_aim_pressed(_touch_aim or Input.is_action_pressed("aim"))

func _process(delta: float) -> void:
	_recoil_yaw = lerpf(_recoil_yaw, 0.0, clampf(delta * 7.0, 0.0, 1.0))
	_recoil_pitch = lerpf(_recoil_pitch, 0.0, clampf(delta * 7.0, 0.0, 1.0))
	rotation.y = _yaw + _recoil_yaw
	camera_pivot.rotation.x = _pitch + _recoil_pitch

func _unhandled_input(event: InputEvent) -> void:
	if event is InputEventMouseButton:
		var mouse_button: InputEventMouseButton = event as InputEventMouseButton
		if mouse_button.button_index == MOUSE_BUTTON_RIGHT and mouse_button.pressed:
			Input.mouse_mode = Input.MOUSE_MODE_CAPTURED
	elif event is InputEventKey:
		var key_event: InputEventKey = event as InputEventKey
		if key_event.keycode == KEY_ESCAPE and key_event.pressed:
			Input.mouse_mode = Input.MOUSE_MODE_VISIBLE
	elif event is InputEventMouseMotion and Input.mouse_mode == Input.MOUSE_MODE_CAPTURED:
		var motion: InputEventMouseMotion = event as InputEventMouseMotion
		add_look_delta(motion.relative, true)

func _get_move_input() -> Vector2:
	if _touch_move.length_squared() > 0.01:
		return _touch_move
	return Input.get_vector("move_left", "move_right", "move_forward", "move_back")

func _update_camera(delta: float, input_vector: Vector2, moving: bool, sprinting: bool, tactical_sprinting: bool) -> void:
	var desired_height: float = 1.55
	if _is_crouched:
		desired_height = 1.05
	if _is_sliding:
		desired_height = 0.78
	camera_pivot.position.y = lerpf(camera_pivot.position.y, desired_height, clampf(delta * 10.0, 0.0, 1.0))
	var capsule: CapsuleShape3D = collision_shape.shape as CapsuleShape3D
	if capsule != null:
		var desired_capsule_height: float = 1.8 if not _is_crouched and not _is_sliding else (1.25 if _is_crouched else 0.95)
		capsule.height = lerpf(capsule.height, desired_capsule_height, clampf(delta * 9.0, 0.0, 1.0))
		collision_shape.position.y = capsule.height * 0.5
	var bob_speed: float = 8.0
	if sprinting:
		bob_speed = 12.5
	if moving and is_on_floor() and not _is_sliding:
		_headbob_time += delta * bob_speed
	else:
		_headbob_time = lerpf(_headbob_time, 0.0, clampf(delta * 5.0, 0.0, 1.0))
	camera.position.y = sin(_headbob_time) * (0.018 if sprinting else 0.012) if moving else 0.0
	var target_tilt: float = -input_vector.x * 0.045
	camera_pivot.rotation.z = lerpf(camera_pivot.rotation.z, target_tilt, clampf(delta * 7.0, 0.0, 1.0))
	var target_fov: float = 75.0
	if sprinting:
		target_fov = 82.0 if tactical_sprinting else 79.0
	if weapon.get_ads_blend() > 0.01:
		target_fov = lerpf(target_fov, weapon.get_ads_fov(), weapon.get_ads_blend())
	camera.fov = lerpf(camera.fov, target_fov, clampf(delta * 6.0, 0.0, 1.0))

func add_look_delta(delta_pixels: Vector2, from_mouse: bool = false) -> void:
	var sensitivity: float = mouse_sensitivity if from_mouse else touch_sensitivity
	sensitivity *= lerpf(1.0, 0.48, weapon.get_ads_blend())
	_yaw -= delta_pixels.x * sensitivity
	_pitch = clampf(_pitch - delta_pixels.y * sensitivity, deg_to_rad(-82.0), deg_to_rad(82.0))
	weapon.add_sway(delta_pixels)
	rotation.y = _yaw + _recoil_yaw
	camera_pivot.rotation.x = _pitch + _recoil_pitch

	if sprinting and is_on_floor() and _slide_cooldown_left <= 0.0:

	else:
		_is_crouched = not _is_crouched

	if _slide_direction.length_squared() < 0.01:
		_slide_direction = -global_transform.basis.z
	_is_sliding = true

func _input(event: InputEvent) -> void:
	if event is InputEventScreenTouch:
		var touch: InputEventScreenTouch = event as InputEventScreenTouch
		if touch.pressed:
			if _point_over_controls(touch.position):
				return
			var viewport_size: Vector2 = get_viewport().get_visible_rect().size
			if touch.position.x < viewport_size.x * 0.46:
				if touch.position.y > viewport_size.y * 0.28 and _joystick_finger == -1:
					_joystick_finger = touch.index
					_joystick.begin_touch(touch.index, touch.position)
			else:
				_camera_fingers[touch.index] = true
		else:
			if touch.index == _joystick_finger:
				_joystick.end_touch(touch.index)
				_joystick_finger = -1
				move_changed.emit(Vector2.ZERO)
			_camera_fingers.erase(touch.index)
	elif event is InputEventScreenDrag:
		var drag: InputEventScreenDrag = event as InputEventScreenDrag
		if drag.index == _joystick_finger:
			_joystick.drag_touch(drag.index, drag.position)
		elif _camera_fingers.has(drag.index):
			look_delta.emit(drag.relative)

class_name VirtualJoystick

## Joystick flotante: el dedo crea el centro y el mismo índice lo controla hasta soltar.
@export var radius: float = 92.0
var owner_index: int = -1
var origin: Vector2 = Vector2.ZERO
var knob_position: Vector2 = Vector2.ZERO
var output_vector: Vector2 = Vector2.ZERO
var active: bool = false

func _ready() -> void:
	set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	mouse_filter = Control.MOUSE_FILTER_IGNORE
	z_index = 8

func begin_touch(index: int, screen_position: Vector2) -> void:
	if active:
		return
	owner_index = index
	active = true
	origin = screen_position
	knob_position = origin
	output_vector = Vector2.ZERO
	queue_redraw()

func drag_touch(index: int, screen_position: Vector2) -> void:
	if not active or index != owner_index:
		return
	var delta_position: Vector2 = screen_position - origin
	var magnitude: float = delta_position.length()
	if magnitude > radius:
		delta_position = delta_position.normalized() * radius
	knob_position = origin + delta_position
	output_vector = delta_position / radius
	queue_redraw()

func end_touch(index: int) -> void:
	if not active or index != owner_index:
		return
	owner_index = -1
	active = false
	output_vector = Vector2.ZERO
	queue_redraw()

func get_vector() -> Vector2:
	return output_vector

func _draw() -> void:
	if not active:
		return
	draw_circle(origin, radius, Color(0.06, 0.09, 0.11, 0.34))
	draw_arc(origin, radius, 0.0, TAU, 56, Color(0.85, 0.9, 0.92, 0.55), 3.0, true)
	draw_circle(knob_position, radius * 0.34, Color(0.76, 0.83, 0.84, 0.66))
	draw_arc(knob_position, radius * 0.34, 0.0, TAU, 40, Color(1.0, 1.0, 1.0, 0.8), 2.0, true)

extends Node3D
class_name WeaponSystem

## Rifle hitscan con dispersión, ADS, recoil, sway y recarga por fases.
signal ammo_changed(current: int, reserve: int)
signal hit_confirmed(headshot: bool, killed: bool)
signal recoil_requested(amount_degrees: Vector2)
signal reload_progress(progress: float, empty_reload: bool)

const TRACER_SCRIPT: Script = preload("res://scripts/BulletTracer.gd")
const WEAPON_DATA: Resource = preload("res://data/WeaponData.tres")

enum ReloadPhase { NONE, EJECT, INSERT, CHAMBER }

var weapon_data: WeaponData = WEAPON_DATA as WeaponData
var magazine_rounds: int = 29
var reserve_ammo: int = 120
var chambered: bool = true
var is_reloading: bool = false
var is_aiming: bool = false
var is_sprinting: bool = false
var is_sliding: bool = false
var fire_held: bool = false
var aim_held: bool = false
var reload_phase: ReloadPhase = ReloadPhase.NONE
var reload_phase_time: float = 0.0
var reload_duration_total: float = 1.0
var reload_elapsed: float = 0.0
var _fire_cooldown: float = 0.0
var _time_since_last_shot: float = 10.0
var _shot_pattern_index: int = 0
var _ads_state: bool = false
const RECOIL_HORIZONTAL_PATTERN: Array[float] = [-0.16, 0.24, 0.31, -0.27, 0.12, -0.08, 0.22, -0.19]
const RECOIL_VERTICAL_PATTERN: Array[float] = [0.88, 0.98, 1.08, 1.18, 1.23, 1.12, 1.28, 1.16]
var _aim_blend: float = 0.0
var _weapon_kick: Vector3 = Vector3.ZERO
var _sway: Vector2 = Vector2.ZERO
var _sway_target: Vector2 = Vector2.ZERO
var _model: Node3D
var _magazine_mesh: MeshInstance3D
var _bolt_mesh: MeshInstance3D
var _muzzle: Node3D
var _camera: Camera3D
var _center_ray: RayCast3D
var _player_body: CollisionObject3D
var _animation_player: AnimationPlayer

func _ready() -> void:
	if weapon_data == null:
		weapon_data = WeaponData.new()
	magazine_rounds = maxi(weapon_data.magazine_capacity - 1, 0)
	reserve_ammo = weapon_data.reserve_ammo
	_camera = get_parent() as Camera3D
	_center_ray = _camera.get_node_or_null("BulletRay") as RayCast3D
	var camera_pivot: Node = _camera.get_parent()
	if camera_pivot != null:
		_player_body = camera_pivot.get_parent() as CollisionObject3D
	if _center_ray != null and _player_body != null:
		_center_ray.add_exception(_player_body)
	_build_weapon_model()
	_build_reload_animations()
	_emit_ammo()

func _process(delta: float) -> void:
	_fire_cooldown = maxf(_fire_cooldown - delta, 0.0)
	_time_since_last_shot += delta
	_sway = _sway.lerp(_sway_target, clampf(delta * 8.0, 0.0, 1.0))
	_sway_target = _sway_target.lerp(Vector2.ZERO, clampf(delta * 3.5, 0.0, 1.0))
	_aim_blend = move_toward(_aim_blend, 1.0 if aim_held and not is_reloading and not is_sprinting else 0.0, delta * 7.0)
	var aiming_now: bool = _aim_blend >= 0.85
	if aiming_now != _ads_state:
		_ads_state = aiming_now
		ads_state_changed.emit(_ads_state)
	_weapon_kick = _weapon_kick.lerp(Vector3.ZERO, clampf(delta * 8.0, 0.0, 1.0))
	_update_weapon_pose(delta)
	if fire_held and _fire_cooldown <= 0.0 and not is_reloading and not is_sprinting and not is_sliding:
		_fire_once()
	if is_reloading:
		_update_reload(delta)

func set_fire_pressed(pressed: bool) -> void:
	fire_held = pressed

func set_aim_pressed(pressed: bool) -> void:
	aim_held = pressed

func set_sprinting(tactical: bool, sliding: bool) -> void:
	is_sprinting = tactical
	is_sliding = sliding

func add_sway(delta_pixels: Vector2) -> void:
	_sway_target.x = clampf(_sway_target.x - delta_pixels.x * 0.00065, -0.075, 0.075)
	_sway_target.y = clampf(_sway_target.y + delta_pixels.y * 0.00055, -0.06, 0.06)

func get_ads_blend() -> float:
	return _aim_blend

func get_ads_fov() -> float:
	return weapon_data.ads_fov

func get_total_ammo() -> int:
	return magazine_rounds + (1 if chambered else 0)

func get_reserve_ammo() -> int:
	return reserve_ammo

func request_reload() -> void:
	if is_reloading or reserve_ammo <= 0:
		return
	if chambered and magazine_rounds >= weapon_data.magazine_capacity:
		return
	if not chambered and magazine_rounds >= weapon_data.magazine_capacity:
		return
	is_reloading = true
	fire_held = false
	aim_held = false
	_aim_blend = 0.0
	reload_elapsed = 0.0
	reload_phase = ReloadPhase.EJECT
	var empty_reload: bool = not chambered
	reload_phase_time = 0.34 if empty_reload else 0.25
	reload_duration_total = 1.39 if empty_reload else 0.75
	if _animation_player != null:
		_animation_player.play("empty_reload" if empty_reload else "tactical_reload")
	reload_progress.emit(0.0, empty_reload)

func _update_reload(delta: float) -> void:
	reload_elapsed += delta
	reload_phase_time -= delta
	var empty_reload: bool = not chambered
	var progress_value: float = clampf(reload_elapsed / maxf(reload_duration_total, 0.01), 0.0, 1.0)
	reload_progress.emit(progress_value, empty_reload)
	if reload_phase_time > 0.0:
		return
	match reload_phase:
		ReloadPhase.EJECT:
			reload_phase = ReloadPhase.INSERT
			reload_phase_time = 0.50 if empty_reload else 0.50
		ReloadPhase.INSERT:
			if empty_reload:
				reload_phase = ReloadPhase.CHAMBER
				reload_phase_time = 0.55
			else:
				_finish_reload()
		ReloadPhase.CHAMBER:
			_finish_reload()

func _finish_reload() -> void:
	var capacity: int = weapon_data.magazine_capacity
	if chambered:
		var needed: int = maxi(capacity - magazine_rounds, 0)
		var loaded: int = mini(needed, reserve_ammo)
		magazine_rounds += loaded
		reserve_ammo -= loaded
	else:
		var loaded_empty: int = mini(capacity, reserve_ammo)
		magazine_rounds = loaded_empty
		reserve_ammo -= loaded_empty
		if magazine_rounds > 0:
			magazine_rounds -= 1
			chambered = true
	if _animation_player != null and _animation_player.is_playing():
		_animation_player.stop()
	if _magazine_mesh != null:
		_magazine_mesh.position = Vector3(0.0, -0.20, 0.04)
	if _bolt_mesh != null:
		_bolt_mesh.position = Vector3(0.0, 0.055, 0.13)
	is_reloading = false
	reload_phase = ReloadPhase.NONE
	reload_progress.emit(1.0, false)
	_emit_ammo()

func _fire_once() -> void:
	if not chambered:
		if magazine_rounds <= 0:
			request_reload()
			return
		magazine_rounds -= 1
		chambered = true
	if not chambered:
		return
	chambered = false
	if magazine_rounds > 0:
		magazine_rounds -= 1
		chambered = true
	_fire_cooldown = 60.0 / maxf(weapon_data.rounds_per_minute, 1.0)
	var ads_factor: float = _aim_blend
	var spread_degrees: float = lerpf(weapon_data.hip_spread_degrees, weapon_data.ads_spread_degrees, ads_factor)
	var spread_radians: float = deg_to_rad(spread_degrees)
	var random_x: float = randf_range(-spread_radians, spread_radians)
	var random_y: float = randf_range(-spread_radians, spread_radians)
	var camera_transform: Transform3D = _camera.global_transform
	var direction: Vector3 = (-camera_transform.basis.z + camera_transform.basis.x * tan(random_x) + camera_transform.basis.y * tan(random_y)).normalized()
	var ray_origin: Vector3 = _camera.global_position
	var ray_end: Vector3 = ray_origin + direction * 90.0
	var impact_position: Vector3 = ray_end
	var impact_normal: Vector3 = -direction
	var hit_collider: Object = null
	if _center_ray != null:
		_center_ray.target_position = Vector3(tan(random_x), tan(random_y), -1.0).normalized() * 90.0
		_center_ray.force_raycast_update()
		if _center_ray.is_colliding():
			impact_position = _center_ray.get_collision_point()
			impact_normal = _center_ray.get_collision_normal()
			hit_collider = _center_ray.get_collider()
	else:
		var query: PhysicsRayQueryParameters3D = PhysicsRayQueryParameters3D.create(ray_origin, ray_end)
		query.collision_mask = 3
		query.collide_with_areas = true
		query.collide_with_bodies = true
		if _player_body != null:
			var excluded_rids: Array[RID] = [_player_body.get_rid()]
			query.exclude = excluded_rids
		var result: Dictionary = get_world_3d().direct_space_state.intersect_ray(query)
		if not result.is_empty():
			impact_position = Vector3(result["position"])
			impact_normal = Vector3(result["normal"])
			hit_collider = result["collider"] as Object
	if hit_collider is Area3D:
		var hit_area: Area3D = hit_collider as Area3D
		var enemy_node: Node = hit_area.get_parent()
		while enemy_node != null and not enemy_node.has_method("take_damage"):
			enemy_node = enemy_node.get_parent()
		if enemy_node != null:
			var headshot: bool = bool(hit_area.get_meta("headshot", false))
			var dealt_damage: float = weapon_data.damage * (weapon_data.headshot_multiplier if headshot else 1.0)
			var killed: bool = bool(enemy_node.call("take_damage", dealt_damage, headshot))
			hit_confirmed.emit(headshot, killed)
	if hit_collider != null:
		_spawn_impact(impact_position, impact_normal, hit_collider is StaticBody3D)
	if _muzzle != null and randf() < 0.42:
		var tracer: BulletTracer = TRACER_SCRIPT.new() as BulletTracer
		get_tree().current_scene.add_child(tracer)
		tracer.configure(_muzzle.global_position, impact_position)
	if _time_since_last_shot > 0.24:
		_shot_pattern_index = 0
	var pattern_index: int = _shot_pattern_index % RECOIL_VERTICAL_PATTERN.size()
	var horizontal_kick: float = RECOIL_HORIZONTAL_PATTERN[pattern_index] + randf_range(-0.045, 0.045)
	var vertical_kick: float = RECOIL_VERTICAL_PATTERN[pattern_index] + randf_range(-0.07, 0.07)
	if ads_factor > 0.5:
		vertical_kick *= 0.78
		horizontal_kick *= 0.75
	_shot_pattern_index += 1
	_time_since_last_shot = 0.0
	recoil_requested.emit(Vector2(horizontal_kick, vertical_kick))
	_weapon_kick.z = 0.12
	_weapon_kick.y = -0.018
	_emit_ammo()

func _emit_ammo() -> void:

func _update_weapon_pose(delta: float) -> void:
	if _model == null:
		return
	var hip_position: Vector3 = Vector3(0.34, -0.27, -0.54)
	var ads_position: Vector3 = Vector3(0.0, -0.12, -0.37)

	if is_sprinting:
		target_position = target_position.lerp(sprint_position, 0.88)
	if is_sliding:
		target_position.y -= 0.12
	_model.position = _model.position.lerp(target_position + Vector3(_sway.x, _sway.y, 0.0) + _weapon_kick, clampf(delta * 12.0, 0.0, 1.0))
	var target_rotation: Vector3 = Vector3(-_sway.y * 0.35, _sway.x * 0.4, -_sway.x * 0.65)
	if is_sprinting:
		target_rotation.z += 0.28
	_model.rotation = _model.rotation.lerp(target_rotation, clampf(delta * 10.0, 0.0, 1.0))

	_health_display_left = maxf(_health_display_left - delta, 0.0)
	if _health_bar_root != null:
		_health_bar_root.visible = _health_display_left > 0.0
	if not is_instance_valid(_target):
		var candidate := get_tree().get_first_node_in_group("player")
		if candidate is PlayerController:
			_target = candidate as PlayerController

	var desired := Vector3.ZERO
	var moving_amount := 0.0
	var target_visible := false
	var distance_to_target := 999.0
	if is_instance_valid(_target):
		var offset := _target.global_position - global_position
		offset.y = 0.0
		distance_to_target = offset.length()
		if distance_to_target <= detection_range:
			target_visible = _has_line_of_sight()
		if target_visible:
			_strafe_phase += delta
			var toward := offset.normalized()
			var side := Vector3(-toward.z, 0.0, toward.x)
			if distance_to_target > 10.0:
				desired = toward
			elif distance_to_target < 6.0:
				desired = -toward + side * sin(_strafe_phase * 1.7) * 0.35
			else:
				desired = side * sin(_strafe_phase * 1.35)
			if distance_to_target <= attack_range and _fire_cooldown <= 0.0:
				_fire_at_player()
				_fire_cooldown = randf_range(0.9, 1.35)
			var look_target := Vector3(_target.global_position.x, global_position.y, _target.global_position.z)
			if global_position.distance_squared_to(look_target) > 0.01:
				var wanted_yaw := atan2(-(look_target.x - global_position.x), -(look_target.z - global_position.z))
				rotation.y = lerp_angle(rotation.y, wanted_yaw, clampf(delta * 5.0, 0.0, 1.0))
			moving_amount = desired.length()
		else:
			desired = _patrol_step()
			moving_amount = desired.length()
	else:
		desired = _patrol_step()
		moving_amount = desired.length()

	if is_on_floor():
		velocity.y = -0.35
	else:
		velocity.y -= 9.8 * delta
	var speed := combat_speed if target_visible else patrol_speed
	velocity.x = move_toward(velocity.x, desired.x * speed, delta * 7.0)
	velocity.z = move_toward(velocity.z, desired.z * speed, delta * 7.0)
	move_and_slide()
	_animate_swat(delta, moving_amount)

func _patrol_step() -> Vector3:
	var offset_x := position.x - _patrol_origin_x
	if offset_x > 3.5:
		_patrol_direction = -1.0
	elif offset_x < -3.5:
		_patrol_direction = 1.0
	return Vector3(_patrol_direction, 0.0, 0.0)

func _has_line_of_sight() -> bool:
	if not is_instance_valid(_target):
		return false
	var from := global_position + Vector3.UP * 1.45
	var to := _target.global_position + Vector3.UP * 1.15
	var query := PhysicsRayQueryParameters3D.create(from, to)
	query.collision_mask = 1
	query.exclude = [get_rid()]
	var hit := get_world_3d().direct_space_state.intersect_ray(query)
	return not hit.is_empty() and hit.get("collider") == _target

func _fire_at_player() -> void:
	if not is_instance_valid(_target):
		return
	var start := global_position + Vector3.UP * 1.34 + (-global_transform.basis.z * 0.42)
	var finish := _target.global_position + Vector3.UP * 1.12
	var tracer := BulletTracer.new()
	get_tree().current_scene.add_child(tracer)
	tracer.configure(start, finish)
	_muzzle_flash.visible = true
	_muzzle_flash.scale = Vector3.ONE * randf_range(0.7, 1.2)
	var flash_tween := create_tween()
	flash_tween.tween_interval(0.055)
	flash_tween.tween_callback(func() -> void:
		if is_instance_valid(_muzzle_flash):
			_muzzle_flash.visible = false
	)
	_target.take_damage(randf_range(5.5, 8.5))

func take_damage(amount: float, headshot: bool) -> bool:
	if not _alive:
		return false
	current_health = maxf(current_health - amount, 0.0)
	_health_display_left = 1.8
	_update_health_display()
	if headshot:
		_play_headshot_sound()
	if current_health <= 0.0:
		_die()
		return true
	return false

func _die() -> void:
	_alive = false
	visible = false
	velocity = Vector3.ZERO
	if _head_area != null:
		_head_area.monitorable = false
	if _torso_area != null:
		_torso_area.monitorable = false
	await get_tree().create_timer(respawn_delay).timeout
	if not is_inside_tree():
		return
	current_health = max_health
	global_position = Vector3(_patrol_origin_x, 0.0, _patrol_origin_z)
	_alive = true
	visible = true
	if _head_area != null:
		_head_area.monitorable = true
	if _torso_area != null:
		_torso_area.monitorable = true
	_update_health_display()

func _update_health_display() -> void:
	if _health_bar_fill != null:
		var ratio := clampf(current_health / maxf(max_health, 1.0), 0.0, 1.0)
		_health_bar_fill.scale.x = maxf(ratio, 0.01)
		_health_bar_fill.position.x = -0.43 * (1.0 - ratio)
	if _health_label != null:
		_health_label.text = str(int(ceil(current_health))) + " HP"
	if _health_bar_root != null:
		_health_bar_root.visible = true
	_health_display_left = 1.8

func _animate_swat(delta: float, movement: float) -> void:
	var moving := clampf(movement, 0.0, 1.0)
	_motion_phase += delta * (2.0 + moving * 7.0)
	var swing := sin(_motion_phase) * 0.42 * moving
	if _left_leg != null:
		_left_leg.rotation.x = swing
	if _right_leg != null:
		_right_leg.rotation.x = -swing
	if _left_arm != null:
		_left_arm.rotation.x = -swing * 0.35 - 0.18
	if _right_arm != null:
		_right_arm.rotation.x = swing * 0.25 - 0.22

func _play_headshot_sound() -> void:
	var player := AudioStreamPlayer3D.new()
	player.stream = load("res://assets/audio/headshot.wav") as AudioStream
	player.volume_db = -8.0
	player.max_distance = 24.0
	add_child(player)
	player.finished.connect(player.queue_free)
	player.play()

func _material(color: Color, metalness: float = 0.08) -> StandardMaterial3D:

	material.albedo_color = color
	material.roughness = 0.8
	material.metallic = metalness
	return material

	var material := _material(color, 0.0)
	material.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
	material.emission_enabled = true
	material.emission = color
	material.emission_energy_multiplier = 1.8
	return material
